# Additional Districts 2.1.0-beta.7 / Stellaris 4.5.2
ad_add_development_government_district = {
	if = {
		limit = {
			NOT = {
				has_deposit = d_ad_government_district
			}
			ad_development_slots_full = no
		}
		add_deposit = d_ad_government_district
	}
}
ad_remove_development_government_district = {
	if = {
		limit = {
			has_deposit = d_ad_government_district
			ad_can_remove_development_government_district = yes
		}
		remove_deposit = d_ad_government_district
	}
}
