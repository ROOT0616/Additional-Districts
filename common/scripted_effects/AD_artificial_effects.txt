# Additional Districts 2.1.0-beta.4 / Stellaris 4.5.2
ad_add_artificial_sky_cities = {
	if = {
		limit = {
			NOT = {
				has_deposit = d_ad_artificial_sky_cities
			}
		}
		if = {
			limit = {
				count_deposits = {
					type = d_ad_sky_cities
					count = 0
				}
			}
			while = {
				count = 2
				add_deposit = d_ad_artificial_sky_cities
			}
		}
		else_if = {
			limit = {
				count_deposits = {
					type = d_ad_sky_cities
					count = 1
				}
			}
			while = {
				count = 1
				add_deposit = d_ad_artificial_sky_cities
			}
		}
	}
}
ad_remove_artificial_sky_cities = {
	if = {
		limit = {
			ad_can_remove_artificial_sky_cities = yes
		}
		while = {
			limit = {
				has_deposit = d_ad_artificial_sky_cities
			}
			remove_deposit = d_ad_artificial_sky_cities
		}
	}
}
