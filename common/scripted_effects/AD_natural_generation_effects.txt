ad_ng_add_archaeological_cities = {
	if = {
		limit = {
			OR = {
				has_planet_flag = ad_ng_secondary
				has_planet_flag = ad_ng_rare
			}
		}
		if = {
			limit = {
				planet_size >= 20
			}
			random_list = {
				50 = {
					add_deposit = d_ad_archaeological_cities
				}
				50 = {
					while = {
						count = 2
						add_deposit = d_ad_archaeological_cities
					}
				}
			}
		}
		else = {
			add_deposit = d_ad_archaeological_cities
		}
	}
	else = {
		if = {
			limit = {
				planet_size >= 20
			}
			while = {
				count = 3
				add_deposit = d_ad_archaeological_cities
			}
		}
		else = {
			if = {
				limit = {
					planet_size >= 15
				}
				while = {
					count = 2
					add_deposit = d_ad_archaeological_cities
				}
			}
			else = {
				add_deposit = d_ad_archaeological_cities
			}
		}
		random_list = {
			30 = {
				add_deposit = d_ad_archaeological_cities
			}
			70 = {
			}
		}
	}
}

	if = {
		limit = {
			has_global_flag = ad_ng_v3_enabled
			ad_ng_eligible = yes
			NOT = {
				has_planet_flag = ad_ng_v3_checked
			}
		}
		set_planet_flag = ad_ng_v3_checked
		if = {
			limit = {
				ad_ng_existing_features = no
			}
			random_list = {
				20 = {
				}
				25 = {
					ad_ng_choose_energy = yes
				}
				30 = {
					ad_ng_choose_mining = yes
				}
				25 = {
					modifier = {
						factor = 0
						ad_ng_can_farm = no
					}
					ad_ng_choose_farming = yes
				}
			}
			if = {
				limit = {
					planet_size >= 20
				}
				random_list = {
					35 = {
						set_planet_flag = ad_ng_secondary
						random_list = {
							25 = {
								modifier = {
									factor = 0
									has_planet_flag = ad_ng_used_energy
								}
								ad_ng_choose_energy = yes
							}
							30 = {
								modifier = {
									factor = 0
									has_planet_flag = ad_ng_used_mining
								}
								ad_ng_choose_mining = yes
							}
							25 = {
								modifier = {
									factor = 0
									has_planet_flag = ad_ng_used_farming
								}
								modifier = {
									factor = 0
									ad_ng_can_farm = no
								}
								ad_ng_choose_farming = yes
							}
						}
						remove_planet_flag = ad_ng_secondary
					}
					65 = {
					}
				}
			}
			else = {
				if = {
					limit = {
						planet_size >= 15
					}

						20 = {
							set_planet_flag = ad_ng_secondary
							random_list = {
								25 = {
									modifier = {
										factor = 0
										has_planet_flag = ad_ng_used_energy
									}
									ad_ng_choose_energy = yes
								}
								30 = {
									modifier = {
										factor = 0
										has_planet_flag = ad_ng_used_mining
									}
									ad_ng_choose_mining = yes
								}
								25 = {
									modifier = {
										factor = 0
										has_planet_flag = ad_ng_used_farming
									}
									modifier = {
										factor = 0
										ad_ng_can_farm = no
									}
									ad_ng_choose_farming = yes
								}
							}
							remove_planet_flag = ad_ng_secondary
						}
						80 = {
						}
					}
				}
			}
			random_list = {
				15 = {
