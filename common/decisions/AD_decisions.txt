	enactment_time = 0
	resources = {
		category = decisions
		cost = {
			minerals = 1000
			alloys = 500
			exotic_gases = 100
		}
	}
	potential = {
		habitable_planet = yes
		owner = {
			is_ai = yes
		}
		owner = {
			has_technology = large_scale_air_construction_technology
		}
		NOT = {
			has_deposit = d_ad_artificial_sky_cities
		}
		count_deposits = {
			type = d_ad_sky_cities
			count < 2
		}
	}
	effect = {
		custom_tooltip = decision_ad_large_scale_air_area_construction_effects
		hidden_effect = {
			ad_add_artificial_sky_cities = yes
		}
	}
	ai_weight = {
		weight = 50
		modifier = {
			factor = 10
			owner = {
				AND = {
					has_country_resource = {
						type = alloys
						amount > 500
					}
					has_country_resource = {
						type = minerals
						amount > 1000
					}
					has_country_resource = {
						type = exotic_gases
						amount > 100
					}
				}
			}
		}
		modifier = {
			factor = 10
			free_housing < 500
		}
		modifier = {
			factor = 500
			free_housing < 100
		}
	}
	allow = {
		NOT = {
			has_deposit = d_ad_artificial_sky_cities
		}
	}
