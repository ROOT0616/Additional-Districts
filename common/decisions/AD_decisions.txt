decision_ad_large_scale_air_area_construction = {
	owned_planets_only = yes
	enactment_time = 180
	resources = {
		category = decisions
		cost = {
			minerals = 500
			alloys = 250
			exotic_gases = 50
		}
	}
	potential = {
		is_colony = yes
		owner = {
			has_technology = large_scale_air_construction_technology
		}
		NOT = {
			has_planet_flag = ad_artificial_project
		}
		ad_can_start_artificial_sky_cities = yes
	}
	allow = {
		ad_artificial_candidate_sky_cities = yes
		ad_capacity_for_artificial_one_sky_cities = yes
		OR = {
			NOT = {
				has_planet_flag = ad_artificial_project
			}
			has_planet_flag = ad_artificial_project_sky_cities
		}
	}
	on_queued = {
		set_planet_flag = ad_artificial_project
		set_planet_flag = ad_artificial_project_sky_cities
	}
	on_unqueued = {
		remove_planet_flag = ad_artificial_project
		remove_planet_flag = ad_artificial_project_sky_cities
	}
	effect = {
		custom_tooltip = decision_ad_large_scale_air_area_construction_effects
		remove_planet_flag = ad_artificial_project
		remove_planet_flag = ad_artificial_project_sky_cities
		hidden_effect = {
			ad_add_artificial_one_sky_cities = yes
		}
	}
	ai_weight = {
		weight = 5
		modifier = {
			factor = 0
			ad_ai_can_develop_sky_cities = no
		}
		modifier = {
			factor = 2
			free_housing < 300
		}
		modifier = {
			factor = 4

		}
