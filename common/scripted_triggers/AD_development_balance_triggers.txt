# Additional Districts 2.1.0-beta.6 / Stellaris 4.5.2
ad_artificial_surface_colony = {
	is_colony = yes
	OR = {
		uses_district_set = standard
		uses_district_set = city_world
		uses_district_set = ring_world
		uses_district_set = hive_world
		uses_district_set = machine_world
	}
	NOR = {
		has_modifier = resort_colony
		has_modifier = slave_colony
	}
}
ad_ai_development_economy = {
	is_colony = yes
	free_district_slots > 0
	has_building_construction = no
	OR = {
		free_jobs <= 400
		free_housing < 100
	}
	owner = {
		has_monthly_income = {
			resource = energy
			value >= 0
		}
		has_monthly_income = {
			resource = minerals
			value >= 0
		}
	}
}
ad_capacity_for_artificial_one_sky_cities = {
	OR = {
		AND = {
			count_deposits = {
				type = d_ad_sky_cities
				count = 0
			}
			count_deposits = {
				type = d_ad_artificial_sky_cities
				count < 2
			}
		}
		AND = {
			count_deposits = {
				type = d_ad_sky_cities
				count = 1
			}
			count_deposits = {
				type = d_ad_artificial_sky_cities
				count < 1
			}
		}
	}
}
