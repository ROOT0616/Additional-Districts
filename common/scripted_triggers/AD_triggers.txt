is_too_much_industrial = {
	if = {
		limit = {
			owner = {
				is_ai = yes
			}
		}
		OR = {
			owner = {
				has_monthly_income = {
					resource = consumer_goods
					value >= 50
				}
			}
			num_districts = {
				type = district_industrial
				value >= 10
			}
		}
	}
}
